module PE (
    input  wire [3:0] in,
    output reg  [1:0] out,
    output reg        valid
);

    always @(*) begin
        // Check if no bits are set
        if (in == 4'b0000) begin
            valid = 1'b0;
            out   = 2'b00;
        end else begin
            valid = 1'b1;
            // Matches top-down from MSB to LSB
            casez (in)
                4'b1??? : out = 2'b11; // bit 3 set
                4'b01?? : out = 2'b10; // bit 2 set
                4'b001? : out = 2'b01; // bit 1 set
                4'b0001 : out = 2'b00; // bit 0 set
                default : out = 2'b00;
            endcase
        end
    end

endmodule
